const SHEET_NAME = "Enrollments";

// Put your Google Drive folder ID here
const FOLDER_ID = "1Kr5dRFU9uzpR8fisa2aO5cBo_h0qFDYb";


// ========================================
// DO POST
// ========================================

function doPost(e) {

  try {

    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse(false, "No data received.");
    }

    const data = JSON.parse(e.postData.contents);


    // ========================================
    // GET / CREATE SHEET
    // ========================================

    const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();

    let sheet = spreadsheet.getSheetByName(SHEET_NAME);

    // Create sheet if it does not exist
    if (!sheet) {
      sheet = spreadsheet.insertSheet(SHEET_NAME);
    }


    // ========================================
    // AUTOMATIC HEADERS
    // ========================================

    const headers = [
      "Date Submitted",
      "Picture",
      "Term",
      "LRN",
      "First Name",
      "Middle Name",
      "Last Name",
      "Gender",
      "Birthday",
      "Class Code",
      "Section",
      "Adviser"
    ];


    // Add headers automatically if sheet is empty
    if (sheet.getLastRow() === 0) {

      sheet.getRange(1, 1, 1, headers.length)
        .setValues([headers]);

      sheet.getRange(1, 1, 1, headers.length)
        .setFontWeight("bold");
    }


    // ========================================
    // GET FORM DATA
    // ========================================

    const term = data.term || "";
    const LRN = data.LRN || "";
    const FirstName = data.FirstName || "";
    const MiddleName = data.MiddleName || "";
    const LastName = data.LastName || "";
    const gender = data.gender || "";
    const birthday = data.birthday || "";
    const Classcode = data.Classcode || "";
    const section = data.section || "";
    const adviser = data.adviser || "";


    // ========================================
    // UPLOAD PICTURE
    // ========================================

    let pictureURL = "";

    if (data.picture) {

      const folder = DriveApp.getFolderById(FOLDER_ID);

      const base64Data = data.picture.split(",")[1];

      const bytes = Utilities.base64Decode(base64Data);

      const mimeType = data.pictureType || "image/jpeg";

      const blob = Utilities.newBlob(
        bytes,
        mimeType,
        data.pictureName || "student_picture"
      );

      const file = folder.createFile(blob);

      pictureURL = file.getUrl();
    }


    // ========================================
    // DATE AND TIME
    // ========================================

    const dateSubmitted = new Date();


    // ========================================
    // SAVE DATA
    // ========================================

    sheet.appendRow([
      dateSubmitted,
      pictureURL,
      term,
      LRN,
      FirstName,
      MiddleName,
      LastName,
      gender,
      birthday,
      Classcode,
      section,
      adviser
    ]);


    // ========================================
    // SUCCESS
    // ========================================

    return jsonResponse(
      true,
      "Enrollment submitted successfully!"
    );


  } catch (error) {

    console.error(error);

    return jsonResponse(
      false,
      "Error: " + error.message
    );
  }
}


// ========================================
// JSON RESPONSE
// ========================================

function jsonResponse(success, message) {

  return ContentService
    .createTextOutput(
      JSON.stringify({
        success: success,
        message: message
      })
    )
    .setMimeType(ContentService.MimeType.JSON);
}
